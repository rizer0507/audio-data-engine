# 前置：工序一已冻结 Release（有 RELEASE_ID）

cd /data2/data-cp/lizi/audio-data-engine
source /data2/data-cp/zk/env_hub/miniconda312_qw3tts/bin/activate

export BATCH=mt3000
export RELEASE_ID=ds_${BATCH}_v1
export TRAIN_RECIPE=/data2/data-cp/lizi/trainer/recipes/qwen_sft_v3.yaml
export TRAIN_COMMAND="python /data2/data-cp/lizi/trainer/train.py"
export CANDIDATE_MODEL_ID=qwen_sft_${BATCH}_v1
export CANDIDATE_CHECKPOINT=/data2/data-cp/zcl/models/${CANDIDATE_MODEL_ID}

# 1) 外部训练 + 注册 candidate
audio-data training run \
  --release "$RELEASE_ID" \
  --recipe "$TRAIN_RECIPE" \
  --command "$TRAIN_COMMAND" \
  --checkpoint "$CANDIDATE_CHECKPOINT" \
  --model-id "$CANDIDATE_MODEL_ID" \
  --base-model qwen_asr

audio-data model show "$CANDIDATE_MODEL_ID"
