# 详解：docs/07-操作手册/045-入库VAD多路ASR独立DNSMOS自由标签.md
# 台账：单条流水线执行命令.txt §0c
# 须在 tmux 内跑；ASR/vLLM 服务先起好；DNSMOS 模型路径见 configs/quality/dnsmos_p835.yaml

# 进入工程
cd /data2/data-cp/lizi/audio-data-engine
# 激活环境
source /data2/data-cp/zk/env_hub/miniconda312_qw3tts/bin/activate
# 限制原生线程，避免和进程并发打架
export OMP_NUM_THREADS=1
# 同上
export MKL_NUM_THREADS=1

# 批次名（只改这里和 SOURCE_DIR）
export BATCH=20260813-n-1
# 原始音频目录
export SOURCE_DIR=/data2/data-cp/datasets/asr_datasets/fenshen/20260813/1
# FSMN VAD：服务器原始权重（只读资产）
export FSMN_VAD_SOURCE_MODEL_DIR=/data2/data-cp/xijujun/models/fsmn-vad
# 本地运行目录（须已用 prepare 覆盖完整 fenshen config；必须含 config.yaml、configuration.json、model.pt、am.mvn）
# 不能是 FunASR 默认 800/0.6/60000；生产会校验 YAML 与加载后 vad_opts。VAD 语义见 049（v8）。
# 首次准备（每机一次即可）：
# python scripts/prepare_fsmn_vad_runtime.py \
#   --source-model-dir "$FSMN_VAD_SOURCE_MODEL_DIR" \
#   --fenshen-config configs/vad/fenshen_vad_config.yaml \
#   --runtime-dir "$PWD/var/models/fsmn-vad-fenshen-v1"
export FSMN_VAD_RUNTIME_DIR="$PWD/var/models/fsmn-vad-fenshen-v1"
# Qwen vLLM 地址（按部署改，勿照抄）
export QWEN_ASR_API_BASE=http://127.0.0.1:5555
# Qwen API Key（无则 dummy）
export QWEN_ASR_API_KEY=dummy
# GLM vLLM 地址（按部署改）
export GLM_ASR_API_BASE=http://127.0.0.1:5570

# ---------- 1) 入库（单独产生 ingested_$BATCH.parquet）----------
# 统一场景识别 + 父源登记 + 16k mono 规范化 → datasets/stage1/ingested/ingested_$BATCH.parquet
# 同目录可混放：fenshen*（分身 16k）、shenpi_ai-/shepi_ai-（审批；后缀含大写 V/P=面谈 8k，否则外呼 16k）
# 写入 labels.audio_scene=fenshen|shenpi_interview|shenpi_outbound|unknown（展开为 label_audio_scene）
# 历史错误审批入库须换新批次名从原始 PCM 重跑本步，不能只改标签
audio-data pipeline run pipelines/audio_ingest.yaml --source-name "$BATCH" --source-dir "$SOURCE_DIR"

# ---------- 2a) 分身 VAD（FSMN v9；只处理 fenshen / 历史兼容缺标记分身）----------
# 输入 ingested_$BATCH.parquet → 输出 datasets/stage1/segments/segments_$BATCH.parquet
audio-data pipeline run pipelines/vad_segment_batch.yaml --source-name "$BATCH"

# ---------- 2b) 审批 VAD（HTTP FireRedVAD；只处理面谈/外呼；默认产物 -shenpi）----------
# 部署地址按环境配置；无服务时可 AUDIO_ENGINE_SHENPI_VAD_MOCK=1 做替身
# export SHENPI_VAD_API_BASE='http://your-vad-host:8094'
# 输入同一份 ingested_$BATCH → 输出 datasets/stage1/segments/segments_$BATCH-shenpi.parquet
# audio-data pipeline run pipelines/vad_segment_shenpi_batch.yaml --source-name "$BATCH"

# ---------- 2.5) 按 AI 话术标签剔除「接洽（身份核实）」（可选；有 xlsx 时做）----------
# 输入：segments_$BATCH.parquet + LABEL_XLSX（关注列：file_id、AI话术分类标签）
# 匹配键：切段父源 labels.source_audio_id / label_source_audio_id ↔ xlsx.file_id
#         （切段后顶层 id 已是 seg_...，不能拿来对 file_id）
# 流程：脚本先打印列核对与命中统计，确认字段对齐后再删除该标签父源的全部切段
# 产物：全量备份 segments_$BATCH_vad_all.parquet；过滤结果写回 segments_$BATCH.parquet
#       后续 ASR/DNSMOS 用 --source-name 会直接读过滤后的 segments_
# LABEL_XLSX 按实际路径改；本机示例：data/raw/分身13-14日志_合并清洗数据_已打标签.xlsx
export LABEL_XLSX="$PWD/data/raw/分身13-14日志_合并清洗数据_已打标签.xlsx"
# 只看匹配报告、不写盘：
# python scripts/filter_segments_drop_identity_verify.py --source-name "$BATCH" --xlsx "$LABEL_XLSX" --dry-run
python scripts/filter_segments_drop_identity_verify.py --source-name "$BATCH" --xlsx "$LABEL_XLSX"

# ---------- 3) ASR 与 DNSMOS（可并行；都读 segments_）----------
# 分身：VAD_BATCH="$BATCH"；审批：VAD_BATCH="${BATCH}-shenpi"
# export VAD_BATCH="$BATCH"
# export VAD_BATCH="${BATCH}-shenpi"
# ASR 落盘已含转写清洗（048：去控制标签；GLM/SenseVoice 用 zh_asr_v1 去标点；
# Qwen 用 zh_asr_keep_punct_v1 保留标点 + blank_exact，热词仍按无标点 plain 匹配）
# transcripts.*.text 已是清洗正文；原串在 extra.raw_text。不必再为 join/tag 先跑 asr_metric_pipeline。
# label_audio_scene 随片段保留到下游，规则不依赖改写该字段
# Qwen 一路 → datasets/stage1/asr/qwen1_asr_$VAD_BATCH.parquet
audio-data pipeline run pipelines/qwen_asr_batch.yaml --source-name "${VAD_BATCH:-$BATCH}" --asr-run qwen1
# GLM 一路 → datasets/stage1/asr/glm1_asr_$VAD_BATCH.parquet
audio-data pipeline run pipelines/glm_asr_batch.yaml --source-name "${VAD_BATCH:-$BATCH}" --asr-run glm1
# 独立 DNSMOS → datasets/stage1/quality/dnsmos_$VAD_BATCH.parquet
audio-data pipeline run pipelines/dnsmos_score_batch.yaml --source-name "${VAD_BATCH:-$BATCH}"

# ---------- 4) 证据合并 ----------
# 对齐片段 + ASR + DNSMOS → datasets/stage1/derived/prepared_evidence_$VAD_BATCH.parquet
# 若设了 AUDIO_ENGINE_DNSMOS_MANIFEST，须指向同分支同次片段的 DNSMOS
audio-data pipeline run pipelines/evidence_join.yaml --source-name "${VAD_BATCH:-$BATCH}" --join-manifest qwen1 --join-manifest glm1

# ---------- 5) 自由 tag ----------
# 消费已提交证据 → datasets/stage1/derived/tagged_free_tags_v2_$VAD_BATCH.parquet
audio-data pipeline run pipelines/tag_dataset_free_tags_v2.yaml --source-name "${VAD_BATCH:-$BATCH}" --config configs/tagging/zh_asr_free_tags_v2.yaml

# ---------- 进度 / 续跑（各步独立 runs）----------
# 看某次子流水线进度（把路径换成该步 runs 目录）
audio-data pipeline progress "runs/<run-dir>" --follow
# 某步续跑：对该 YAML 再跑一次，加 --resume "runs/<run-dir>"

# ASR 路次可在各 ASR 命令的 --asr-run 与 evidence_join 的 --join-manifest 里改
