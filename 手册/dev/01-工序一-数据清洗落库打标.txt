工序一：入库 → VAD → ASR∥DNSMOS → 打标（045，分步执行）
================================================================================
# 详解：docs/07-操作手册/045-入库VAD多路ASR独立DNSMOS自由标签.md
# 台账：单条流水线执行命令.txt §0c
# 须在 tmux 内跑；ASR/vLLM 服务先起好；DNSMOS 模型路径见 configs/quality/dnsmos_p835.yaml

# 进入工程
cd /data2/data-cp/lizi/audio-data-engine
# 激活环境
source /data2/data-cp/zk/env_hub/miniconda312_qw3tts/bin/activate
# 限制原生线程，避免和进程并发打架
export OMP_NUM_THREADS=1
# 同上
export MKL_NUM_THREADS=1

# 批次名（只改这里和 SOURCE_DIR）
export BATCH=20260813-n-1
# 原始音频目录
export SOURCE_DIR=/data2/data-cp/datasets/asr_datasets/fenshen/20260813/1
# FSMN VAD：服务器原始权重（只读资产）
export FSMN_VAD_SOURCE_MODEL_DIR=/data2/data-cp/xijujun/models/fsmn-vad
# 本地运行目录（须已用 prepare 覆盖 fenshen config；必须含 config.yaml、configuration.json、model.pt、am.mvn）
# 首次准备（每机一次即可）：
# python scripts/prepare_fsmn_vad_runtime.py \
#   --source-model-dir "$FSMN_VAD_SOURCE_MODEL_DIR" \
#   --fenshen-config configs/vad/fenshen_vad_config.yaml \
#   --runtime-dir "$PWD/var/models/fsmn-vad-fenshen-v1"
export FSMN_VAD_RUNTIME_DIR="$PWD/var/models/fsmn-vad-fenshen-v1"
# Qwen vLLM 地址（按部署改，勿照抄）
export QWEN_ASR_API_BASE=http://127.0.0.1:5555
# Qwen API Key（无则 dummy）
export QWEN_ASR_API_KEY=dummy
# GLM vLLM 地址（按部署改）
export GLM_ASR_API_BASE=http://127.0.0.1:5570

# ---------- 1) 入库（单独产生 ingested_$BATCH.parquet）----------
# 父源登记 + 16k mono 规范化 → datasets/stage1/ingested/ingested_$BATCH.parquet
audio-data pipeline run pipelines/audio_ingest.yaml --source-name "$BATCH" --source-dir "$SOURCE_DIR"

# ---------- 2) VAD 切分（读 ingested_ → 写 segments_）----------
# 输入 ingested_$BATCH.parquet → 输出 datasets/stage1/segments/segments_$BATCH.parquet
audio-data pipeline run pipelines/vad_segment_batch.yaml --source-name "$BATCH"

# ---------- 3) ASR 与 DNSMOS（可并行；都读 segments_）----------
# ASR 落盘已含转写清洗（048：去控制标签/zh_asr_v1 去标点；Qwen 另挂 blank_exact）
# transcripts.*.text 已是可比正文；原串在 extra.raw_text。不必再为 join/tag 先跑 asr_metric_pipeline。
# Qwen 一路 → datasets/stage1/asr/qwen1_asr_$BATCH.parquet
audio-data pipeline run pipelines/qwen_asr_batch.yaml --source-name "$BATCH" --asr-run qwen1
# GLM 一路 → datasets/stage1/asr/glm1_asr_$BATCH.parquet
audio-data pipeline run pipelines/glm_asr_batch.yaml --source-name "$BATCH" --asr-run glm1
# 独立 DNSMOS → datasets/stage1/quality/dnsmos_$BATCH.parquet
audio-data pipeline run pipelines/dnsmos_score_batch.yaml --source-name "$BATCH"

# ---------- 4) 证据合并 ----------
# 对齐片段 + ASR + DNSMOS → datasets/stage1/derived/prepared_evidence_$BATCH.parquet
audio-data pipeline run pipelines/evidence_join.yaml --source-name "$BATCH" --join-manifest qwen1 --join-manifest glm1

# ---------- 5) 自由 tag ----------
# 消费已提交证据 → datasets/stage1/derived/tagged_free_tags_v2_$BATCH.parquet
audio-data pipeline run pipelines/tag_dataset_free_tags_v2.yaml --source-name "$BATCH" --config configs/tagging/zh_asr_free_tags_v2.yaml

# ---------- 进度 / 续跑（各步独立 runs）----------
# 看某次子流水线进度（把路径换成该步 runs 目录）
audio-data pipeline progress "runs/<run-dir>" --follow
# 某步续跑：对该 YAML 再跑一次，加 --resume "runs/<run-dir>"
