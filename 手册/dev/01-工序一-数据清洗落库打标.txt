# 对齐：docs/01-项目架构/工序总览.md · 工序一（①清洗 + ②ASR/聚拢/字准 + ③分拣）
# 详解：手册/dev/00-总详细手册.txt
# 前置：另开终端已起好 Qwen vLLM（例：port 5555）
# 改批次只改 BATCH / SOURCE_DIR

cd /data2/data-cp/lizi/audio-data-engine
source /data2/data-cp/zk/env_hub/miniconda312_qw3tts/bin/activate
export OMP_NUM_THREADS=1
export MKL_NUM_THREADS=1

export BATCH=mt3000
export SOURCE_DIR=/data2/data-cp/lizi/mt3000
export QWEN_ASR_API_BASE=http://127.0.0.1:5555
export QWEN_ASR_API_KEY=dummy
export SENSEVOICE_MODEL_PATH=/data2/data-cp/xijujun/models/SenseVoiceSmall

# 1) 清洗落库
audio-data pipeline run pipelines/data_cleaning_source_A.yaml \
  --source-name "$BATCH" \
  --source-dir "$SOURCE_DIR"

# 2) Qwen 识别（vLLM 已就绪）
curl -fsS http://127.0.0.1:5555/v1/models | python -m json.tool
audio-data pipeline run pipelines/qwen_asr_batch.yaml --source-name "$BATCH"

# 3) SenseVoice 识别
audio-data pipeline run pipelines/sensevoice_asr_batch.yaml --source-name "$BATCH"

# 4) 可选：Kimi 本地 / Doubao（按需取消注释）
# export KIMI_AUDIO_MODEL_PATH=/data2/data-cp/models/kimi-audio
# audio-data pipeline run pipelines/kimi_audio_asr_batch.yaml --source-name "$BATCH"
# audio-data pipeline run pipelines/doubao_asr_batch.yaml --source-name "$BATCH" --asr-run doubao1

# 5) 聚合 + 字准
audio-data pipeline run pipelines/multi_asr_aggregate.yaml \
  --source-name "$BATCH" \
  --join-manifest sensevoice \
  --force
# 若有更多模型：再加 --join-manifest doubao --join-manifest kimi_audio 等

audio-data pipeline run pipelines/asr_metric_pipeline.yaml \
  --source-name "$BATCH" \
  --force

# 6) 分拣
audio-data pipeline run pipelines/classify_dataset.yaml \
  --input-manifest "datasets/manifests/multi_asr_metrics_${BATCH}.parquet" \
  --output-manifest "datasets/manifests/classified_${BATCH}.parquet" \
  --force

# 7) 导出汇总（可先搁置人审）
audio-data review export-summary "datasets/manifests/classified_${BATCH}.parquet" \
  --output "data/exports/summary_${BATCH}.xlsx" \
  --max-rows 20000
