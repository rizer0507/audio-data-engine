前后端启动（服务器内网）
================================================================================
# 说明：内网 Maven 只有 PyPI，没有可用 npm 源；npm ci 会卡死。
# 前端在本机 build，经 U 盘同步 frontend/dist；服务器只装 Python Web 依赖并启动。

# 工程根
cd /data2/data-cp/lizi/audio-data-engine
# conda 环境（与现场一致）
source /data2/data-cp/zk/env_hub/miniconda312_qw3tts/bin/activate
# 限制原生线程
export OMP_NUM_THREADS=1
# 同上
export MKL_NUM_THREADS=1

# 安装/更新 Web 依赖（内网 PyPI）
pip install -e ".[web]" --index-url http://maven.paic.com.cn/repository/pypi/simple --trusted-host maven.paic.com.cn
# 确认静态页已同步（缺则先跑本机 npm run build + U 盘热更新）
test -f frontend/dist/index.html

# 终端1：Web（同源托管 frontend/dist，默认 127.0.0.1:5179）
audio-data-web --config /data2/data-cp/lizi/audio-data-engine/configs/web.yaml --project-root /data2/data-cp/lizi/audio-data-engine
# 终端2：Worker（领取并执行任务）
audio-data-worker --config /data2/data-cp/lizi/audio-data-engine/configs/web.yaml --project-root /data2/data-cp/lizi/audio-data-engine
