单条流水线执行命令
=======================================================
用途：维护生产入口命令；分片/并行/合并由 YAML sharding 自动完成。
改批次：用 --source-name（+ 清洗时的 --source-dir），一般不用改 YAML 路径。
改并行度 / 卡号：改对应 pipelines/*.yaml 的 sharding。
分拣默认：工序一③ = classify_dataset_five_class_v2_2_auto_noise（029）；028/v1 为对照；legacy = asr_anomaly + selection_v3.0。
执行顺序：1 清洗 → 2 ASR（≥3 族双跑 + register）→ 3 prepare/attach → energy/DNSMOS 候选 → 五分类。
         噪声全量 sidecar 仅研究；聚合+字准（2d/2e）与 classify_dataset(v1) 仅为诊断 / legacy。
噪声速查：§2f-noise；正式五类入口：§2f-v3-029。
服务器过夜批跑：手工 vLLM / pipeline 必须进 tmux（见 手册/dev/00 §2.0）；
  Tabby/VS Code 不能保活。stage1 单命令（0a/030）本身后台跑，SSH 断开可继续。

# 必填：authorized_gpus / engine_python / families.qwen.vllm_bin / dnsmos.onnx_path

# 若仍手工起族内 vLLM 对照，则那些 serve 仍须 tmux。
# 台账：手册/dev/05-一条命令多步骤.txt §1

audio-data stage1 check-config --runtime-config configs/stage1/server.yaml

audio-data stage1 run \
  --batch "$BATCH" \
  --source "$SOURCE_DIR" \
  --model qwen=/data2/data-cp/zcl/models/Qwen3-ASR-1___7B \
  --model glm=/data2/data-cp/models/GLM-ASR-Nano-2512 \
  --model sensevoice=/data2/data-cp/xijujun/models/SenseVoiceSmall \
  --runtime-config configs/stage1/server.yaml \
  --gpus "$AUTHORIZED_GPUS"

audio-data stage1 status <job_id> --watch
audio-data stage1 status <job_id> --json
audio-data stage1 wait <job_id>
# wait：reconcile.ok → exit 0；failed → 1；needs_attention → 3
# 提交成功 ≠ 批次成功

audio-data stage1 resume <job_id>
audio-data stage1 retry <job_id> --family glm --run 2 --failed-only
audio-data stage1 retry <job_id> --export-only

# 产物：
#   runs/stage1/jobs/<job_id>/  (state.json / checkpoints / worker.lock / events.jsonl)
#   datasets/stage1/cleaned/cleaned_${BATCH}.parquet
#   datasets/stage1/asr/{qwen,glm,sensevoice}_{1,2}_asr_${BATCH}.parquet
#   datasets/stage1/derived/classified_five_class_v2_2_auto_noise_${BATCH}.parquet
