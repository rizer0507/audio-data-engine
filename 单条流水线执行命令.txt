单条流水线执行命令
=======================================================
用途：维护生产入口命令；分片/并行/合并由 YAML sharding 自动完成。
改批次：用 --source-name（+ 清洗时的 --source-dir），一般不用改 YAML 路径。
改并行度 / 卡号：改对应 pipelines/*.yaml 的 sharding。
执行顺序：1 清洗 → 2a/2b/2c 各模型独立识别 → 2d 聚合 → 2e 字准率 → 导出 xlsx（可选）。


-------------------------------------------------------
0. 进入工程（每次开终端先做）
-------------------------------------------------------
cd /workspace/audio-data-engine
source .venv/bin/activate
export OMP_NUM_THREADS=1
export MKL_NUM_THREADS=1


-------------------------------------------------------
1. 数据切片落库
-------------------------------------------------------
# YAML: pipelines/data_cleaning_source_A.yaml
# 产物: datasets/manifests/cleaned_<source-name>.parquet|.jsonl

audio-data pipeline run pipelines/data_cleaning_source_A.yaml \
  --source-name test_local \
  --source-dir "D:\Work\audio-data-engine\数据集\wav-file"

# 续跑:
# audio-data pipeline run pipelines/data_cleaning_source_A.yaml \
#   --source-name mt3000 --source-dir /path/to/wav_or_pcm \
#   --resume runs/<上次_shards目录>


-------------------------------------------------------
2a. Qwen 全量识别（独立）
-------------------------------------------------------
# → cleaned_<name> → qwen_asr_<name>.parquet
export QWEN_ASR_MODEL_PATH="/home/lizi/Qwen3-ASR-main/data/outputs/0814-1506/checkpoint-1500"

audio-data pipeline run pipelines/qwen_asr_batch.yaml --source-name test_local --force

# 总进度（父进程控制台 + runs/<run>_shards/PROGRESS.log）：
#   [PROGRESS] done=1234/8000 (15.4%) rate=42.30 samples/s ...
# 另开终端查看：
#   tail -f runs/<上次_shards目录>/PROGRESS.log
#   audio-data pipeline progress runs/<上次_shards目录> -f


-------------------------------------------------------
2b. SenseVoice 全量识别（独立，不依赖 Qwen，不算字准率）
-------------------------------------------------------
# → cleaned_<name> → sensevoice_asr_<name>.parquet
export SENSEVOICE_MODEL_PATH=/data2/data-cp/xijujun/models/SenseVoiceSmall

audio-data pipeline run pipelines/sensevoice_asr_batch.yaml --source-name test_local


-------------------------------------------------------
2c. Kimi-Audio 全量识别（独立，vLLM HTTP；不依赖 Qwen/SenseVoice）
-------------------------------------------------------
# 前置：先启动 vLLM 服务（另开终端，常驻）
# YAML: pipelines/kimi_asr_batch.yaml
# 产物: cleaned_<name> → kimi_asr_<name>.parquet
# 并行：改 pipelines/kimi_asr_batch.yaml 的 sharding；
#       改 configs/asr/kimi.yaml 的 concurrency（对接 vLLM --max-num-seqs）

CUDA_VISIBLE_DEVICES=0 \
vllm serve /data2/data-cp/zcl/models/Kimi-Audio-7B-Instruct \
  --host 0.0.0.0 \
  --port 5554 \
  --served-model-name kimi-audio \
  --tensor-parallel-size 1 \
  --trust-remote-code \
  --dtype bfloat16 \
  --max-model-len 4096 \
  --max-num-seqs 2 \
  --limit-mm-per-prompt '{"audio":1}'

# 批跑（vLLM 就绪后）
export KIMI_ASR_API_BASE=http://127.0.0.1:5554
export KIMI_ASR_MODEL=kimi-audio

audio-data pipeline run pipelines/kimi_asr_batch.yaml --source-name test_local

# 续跑:
# audio-data pipeline run pipelines/kimi_asr_batch.yaml \
#   --source-name test_local --resume runs/<上次_shards目录>

# 总进度（父进程控制台 + runs/<run>_shards/PROGRESS.log）：
#   [PROGRESS] done=1234/8000 (15.4%) rate=42.30 samples/s ...
# 另开终端查看：
#   tail -f runs/<上次_shards目录>/PROGRESS.log
#   audio-data pipeline progress runs/<上次_shards目录> -f


-------------------------------------------------------
2c-local. Kimi-Audio 全量识别（独立，本地 kimia_infer；不依赖 Qwen/SenseVoice/vLLM）
-------------------------------------------------------
# YAML: pipelines/kimi_audio_asr_batch.yaml
# 产物: cleaned_<name> → kimi_audio_asr_<name>.parquet
# 并行：两卡 A800，每卡 1 进程，进程内一次一条 generate。
#       改 pipelines/kimi_audio_asr_batch.yaml 的 gpus（默认 [0, 1]）。

export KIMI_AUDIO_MODEL_PATH=/data2/data-cp/models/kimi-audio

# 必须先过探针（退出码 0）再跑 Parquet：
python scripts/probe_kimi_audio.py /path/to/short.wav
python scripts/probe_kimi_audio.py /path/to/wav_folder --limit 8

audio-data pipeline run pipelines/kimi_audio_asr_batch.yaml --source-name test_local

# 续跑:
# audio-data pipeline run pipelines/kimi_audio_asr_batch.yaml \
#   --source-name test_local --resume runs/<上次_shards目录>

# 聚合时 join 名是 kimi_audio（不要写成 kimi）：
# audio-data pipeline run pipelines/multi_asr_aggregate.yaml --source-name test_local \
#   --join-manifest sensevoice --join-manifest kimi_audio


-------------------------------------------------------
2d. 按 id 聚合多模型结果（不推理、不算指标）
-------------------------------------------------------
# base: qwen_asr_<name>；join: YAML 中 model 对应 {model}_asr_<name>
# 产物: multi_asr_aggregate_<name>.parquet

audio-data pipeline run pipelines/multi_asr_aggregate.yaml --source-name test_local

# 显式指定要聚合的模型（可重复）：
# audio-data pipeline run pipelines/multi_asr_aggregate.yaml --source-name mt3000 \
#   --join-manifest sensevoice --join-manifest kimi


-------------------------------------------------------
2e. 字准率 / agreement CER（独立 Metric 流水线）
-------------------------------------------------------
# → multi_asr_aggregate_<name> → multi_asr_metrics_<name>.parquet
# 配置: configs/metrics/gold_agreement.yaml

audio-data pipeline run pipelines/asr_metric_pipeline.yaml --source-name test_local


-------------------------------------------------------
2f. 导出 xlsx（parquet/jsonl 全量保留后按需转换）
-------------------------------------------------------
python scripts/export_multi_asr_xlsx.py \
  --manifest datasets/manifests/multi_asr_aggregate_test_local.parquet \
  --output datasets/exports/multi_asr_aggregate_test_local.xlsx \
  --baseline qwen \
  --models qwen,sensevoice,kimi


-------------------------------------------------------
数据链（source-name=mt3000 示例）
-------------------------------------------------------
--source-dir (wav/pcm)
  → cleaned_mt3000.parquet|.jsonl
  → qwen_asr_mt3000.parquet              # 2a 独立产物
  → sensevoice_asr_mt3000.parquet        # 2b 独立产物
  → kimi_asr_mt3000.parquet              # 2c vLLM 产物
  → kimi_audio_asr_mt3000.parquet        # 2c-local 本地产物
  → multi_asr_aggregate_mt3000.parquet   # 2d 仅 join
  → multi_asr_metrics_mt3000.parquet     # 2e agreement CER
  → (可选) datasets/exports/multi_asr_aggregate_mt3000.xlsx

-------------------------------------------------------
换批次时改哪里（清单）
-------------------------------------------------------
[ ] 命令中的 --source-name / --source-dir
[ ] （可选）pipelines/qwen_asr_batch.yaml、sensevoice_asr_batch.yaml 的 gpus/shards
[ ] （可选）pipelines/kimi_asr_batch.yaml 的 sharding；configs/asr/kimi.yaml 的 concurrency
[ ] （可选）pipelines/kimi_audio_asr_batch.yaml 的 gpus（两卡 A800 默认 [0,1]）
[ ] （可选）KIMI_AUDIO_MODEL_PATH；先跑 scripts/probe_kimi_audio.py
[ ] （可选）multi_asr_aggregate.yaml 的 manifests / --join-manifest
[ ] 本文件中的模型环境变量（含 KIMI_ASR_API_BASE / KIMI_ASR_MODEL / KIMI_AUDIO_MODEL_PATH）
# 一般不再改 resources/manifest.yaml 与 YAML 内 input/output 路径
